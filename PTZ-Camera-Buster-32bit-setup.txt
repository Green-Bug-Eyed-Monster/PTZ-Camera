----------------------------------------------------------
                       Enable SSH
----------------------------------------------------------

// Mount the SD card and create an empty file in it's /boot folder named: 

cd ?/boot

touch ssh


----------------------------------------------------------
                       Setup Wifi
----------------------------------------------------------

// Create you WPA supplicant file, and enter content as follows:

cd ?/boot

nano wpa_supplicant.conf

```
country=<YOUR_COUNRTY_CODE>

ctrl_interface=DIR=/var/run/wpa_supplicant GROUP=netdev
update_config=1
ap_scan=1

network={
    ssid="<YOUR_NETWORK_SSID>"
    psk="<YOUR_NETWORD_PASSWORD>"
    key_mgmt=WPA-PSK
}
```


// Insert the SD card in your Pi, boot and wait until it connects to your network by WiFi.

// Login to the Pi by secure shell (SSH), using the default Raspbian credentials:

username: pi

password: raspberry



----------------------------------------------------------
                     Replace Pi user
----------------------------------------------------------

// Create a new user, assign a password, give the new user 
// the same privileges as pi, then remove pi user:

sudo adduser <YOUR_USERNAME>

sudo usermod -aG sudo <YOUR_USERNAME>


// Logout, then login again using the new user's credentials, then remove the old pi account:

sudo deluser --remove-home pi


// Alter new user group memberships as follows. This will allow your
// new user to access the camera and USB serial ports:

sudo usermod -aG video <YOUR_USERNAME>     // add user to video group

sudo usermod -aG dialout <YOUR_USERNAME>   // add user to dialout group (for /dev/ttyUSB0)

sudo reboot



----------------------------------------------------------
        Remove annoying sudo password requirement 
----------------------------------------------------------

sudo visudo

// Add this line to to *END* of the file:

<YOUR_USERNAME> ALL=(ALL:ALL) NOPASSWD: ALL


// Logout, then log back in.



----------------------------------------------------------
                   Update & Upgrade
----------------------------------------------------------

// Adjust package source lists as follows:

sudo nano /etc/apt/sources.list

// Make sure just this line is available and is uncommented.

deb https://legacy.raspbian.org/raspbian/ buster main contrib non-free rpi


sudo nano /etc/apt/sources.list.d/raspi.list

// Make sure just this line is available and is uncommented.

deb http://archive.raspberrypi.org/debian/ buster main


sudo apt update --allow-releaseinfo-change

sudo apt upgrade

// Go make coffee



----------------------------------------------------------
                      raspi-config
----------------------------------------------------------

sudo raspi-config

Set the Pi's hostname to exactly: camera
Set camera support: enabled
Adjust the GPU memory allocation to: 256MB
Adjust timezone to suit your locale.

----------------------------------------------------------

                      Grow Swapfile

----------------------------------------------------------

// To be able to install Node without it hanging due to lack of memory
// (regardless of the GPU memory allocation), we'll have to expand
// the system's swapfile.

// Install dphys-swapfile

sudo apt update

sudo apt install dphys-swapfile

// Turn swapfile off

sudo dphys-swapfile swapoff

// Increase it's size

sudo nano /etc/dphys-swapfile

CONF_SWAPSIZE=512

// Delete the original swap file and recreate it

sudo dphys-swapfile setup

// Re-enable swapfile

sudo dphys-swapfile swapon

sudo reboot

// Check the new size.

free
                  total        used        free      shared  buff/cache   available
    Mem:         243560       36144      138528        1876       68888      155564
    Swap:        524284           0      524284




----------------------------------------------------------
              Install motion and MotionEye
----------------------------------------------------------

// First install dependencies:

sudo apt-get install ffmpeg libmariadb3 libpq5 libmicrohttpd12 -y


// Install Buster version of motion:

wget https://github.com/Motion-Project/motion/releases/download/release-4.3.2/pi_buster_motion_4.3.2-1_armhf.deb

sudo dpkg -i pi_buster_motion_4.3.2-1_armhf.deb

rm pi_buster_motion_4.3.2-1_armhf.deb



// Install some more dependencies:

sudo apt-get install python-pip python-dev libssl-dev libcurl4-openssl-dev libjpeg-dev libz-dev -y


// Install motioneye, which will automatically pull Python dependencies:

pip install motioneye


// Prepare the configuration directory:

sudo mkdir -p /etc/motioneye

sudo cp /home/<YOUR_USERNAME>/.local/share/motioneye/extra/motioneye.conf.sample /etc/motioneye/motioneye.conf


// Create the media directory:

sudo mkdir -p /var/lib/motioneye


// To automatically run motioneye at startup, write a copy of this
// init script to the systemd folder as follows:

sudo nano /etc/systemd/system/motioneye.service

```
[Unit]
Description=motionEye Server
After=network.target

[Service]
Environment="PYTHONPATH=/home/<YOUR_USERNAME>/.local/lib/python2.7/site-packages"
ExecStart=/home/<YOUR_USERNAME>/.local/bin/meyectl startserver -c /etc/motioneye/motioneye.conf
Restart=on-abort

[Install]
WantedBy=multi-user.target
```


// Backup old motioneye camera control files.

sudo cp /home/<YOUR_USERNAME>/.local/lib/python2.7/site-packages/motioneye/mmalctl.py \
        /home/<YOUR_USERNAME>/.local/lib/python2.7/site-packages/motioneye/mmalctl.py.bak


// Make old motioneye compatible with new vcgencmd command.

nano /home/<YOUR_USERNAME>/.local/lib/python2.7/site-packages/motioneye/mmalctl.py

find and replace:   d = dict(p.split('=', 1) for p in support.split())

with:               d = dict(p.split('=', 1) for p in support.split() if '=' in p)

find and replace:   if d.get('detected') == d.get('supported') == '1':

with:               if d.get('detected', '').rstrip(',') == d.get('supported') == '1':


// Start and enable motioneye so it will be available at next boot:

sudo systemctl daemon-reload

sudo systemctl enable motioneye

sudo systemctl start motioneye

sudo systemctl status motioneye


// Reboot

sudo reboot


// If you ever need to keep motioneye up to date, use these commands:

pip install motioneye --upgrade

sudo systemctl restart motioneye

----------------------------------------------------------
              Install node.js v12.22.12
----------------------------------------------------------

// Install Node Version Manager.

curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.39.3/install.sh | bash


// Logout, then back in.


// Check the nvm version.

nvm -v
  0.39.3


// Install Node.js version 12.22.12.

nvm install 12.22.12


// Create our project's parent folder:

sudo mkdir ~/node.js

cd ~/node.js


// Select the Node.js version to use here.

nvm use 12.22.12                                // creates a package.json file in ~/node.js/
  Now using node v12.22.12 (npm v6.14.16)
  

// Check the Node.js version in use here.

node -v
  v12.22.12
  

// Set a default alias for the Node.js version in use.

nvm alias default 12.22.12





----------------------------------------------------------
          Install the accelstepper-module project
----------------------------------------------------------

// Adjust some permissions that will allow us sftp access 
// to the parent folder, for later fine-tuning:

sudo chmod -R 755 ~/node.js

sudo chown -R <YOUR_USERNAME>:<YOUR_USERNAME> ~/node.js


// Install git:

sudo apt update

sudo apt install git


// Install the project folder and files:

cd ~/node.js
git clone --depth 1 https://github.com/Green-Bug-Eyed-Monster/PTZ-Camera.git .tmp-ptz
cp -r .tmp-ptz/accelstepper-module .
rm -rf .tmp-ptz



// Install some further prerequisites:

sudo apt update

sudo apt install -y build-essential python3 g++ make


// Install project JavaScript module dependencies:

cd ~/node.js/accelstepper-module

npm install      // Ignore numerous warnings


// Test the project webpage served by Node.

node index.js

http://camera.local:3000/

// You should see program debug output in your shell, as joystick inputs are manipulated.

Ctrl-C



// To run our Node application at boot, create another systemd unit:

sudo nano /etc/systemd/system/accelstepper.service

```
[Unit]
Description=AccelStepper Node.js Controller
After=network.target

[Service]
Type=simple
User=<YOUR_USERNAME>
WorkingDirectory=/home/<YOUR_USERNAME>/node.js/accelstepper-module
ExecStart=/home/<YOUR_USERNAME>/.nvm/versions/node/v12.22.12/bin/node /home/<YOUR_USERNAME>/node.js/accelstepper-module/index.js
Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target
```
